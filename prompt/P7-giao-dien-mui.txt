Sample `static/js/session_id.js`:

```javascript
// Sinh và giữ định danh phiên trong localStorage; tuyệt đối không lưu khóa API ở đây.
let tam = null;
const HOP_LE = /^[A-Za-z0-9-]{8,64}$/;

function sinhId() {
  if (globalThis.crypto && typeof globalThis.crypto.randomUUID === "function") {
    return globalThis.crypto.randomUUID();
  }
  return "s-" + Math.random().toString(36).slice(2) + Date.now().toString(36);
}

export function getSessionId() {
  const khoa = "session_id";
  let gia = null;
  try {
    gia = localStorage.getItem(khoa);
  } catch (loi) {
    return tam || (tam = sinhId());
  }
  if (gia !== null && HOP_LE.test(gia)) {
    return gia;
  }
  const moi = sinhId();
  try {
    localStorage.setItem(khoa, moi);
  } catch (loi) {
    tam = moi;
  }
  return moi;
}
```

  const chu = laNguoi ? "primary.contrastText" : "text.primary";

      { sx: { mx: 1 }, "aria-label": laNguoi ? "Biểu tượng người dùng" : "Biểu tượng trợ lý" },

      MaterialUI.Paper,
      {

          p: 1.5,
          maxWidth: "80%",
          backgroundColor: nen,
          color: chu,
          whiteSpace: "pre-wrap",

    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      gui();
    }
  };

  return React.createElement(

```javascript
// Gửi một lượt chat và trả về trạng thái mới cho khung chat.
// Không giữ DOM, không giữ khóa API, chỉ đọc session_id đã sinh ở session_id.js.
import { apiChat } from "./api_chat.js";

export async function sendHandler(trangThai, text) {
  const hienTai = trangThai.history.concat([{ role: "human", content: text }]);
  const kq = await apiChat(trangThai.sessionId, text);
  if (kq.ok) {
    const traLoi = (kq.data && kq.data.reply) || "";
    return { history: hienTai.concat([{ role: "ai", content: traLoi }]), error: "", canSetup: false, setupMessage: "" };
  }
  if (kq.status === 409) {
    return { history: hienTai, error: "", canSetup: true, setupMessage: "Chưa có khóa API, hãy nhập lại." };
  }
  if (kq.status === 401) {
    return { history: hienTai, error: "", canSetup: true, setupMessage: "Khóa API không đúng." };
  }
  const nhan = (kq.data && kq.data.message) || "";
  const gh = kq.status === 0 ? "Không kết nối được máy chủ." : (nhan || "Lỗi máy chủ, thử lại.");
  return { history: hienTai, error: gh, canSetup: false, setupMessage: "" };
}

## 3b. Short glossary

- opencode: a coding helper that runs in a command window, used to generate files in this project.
- API: the programming interface between browser and server, here the five `/api/...` paths declared at B7.
- endpoint: a specific path of the API, for example `POST /api/setup`.
- venv: a virtual Python environment created with `py -m venv .venv`, it isolates installed packages from the real machine.
- pytest: the automated test runner for Python, invoked with `py -m pytest -q`.
- DDD: domain-layered architecture, the one-way dependency arrow rule goes from interface to application then to domain.
- TDD: write the test first, see it red, then write the minimal code to make the test green.
- MUI: the Material UI library for React, here the version 5 UMD loaded from a CDN and used through the `MaterialUI` global variable.
- formal spec: a formal specification that records preconditions PRE, postconditions POST, invariants INV, errors ERR and kill conditions KILL with logic symbols.
- UMD: a library packaging format for the browser, loaded with a `script` tag then read through a global variable.
- ES module: the browser's standard module format, loaded with `type="module"`, uses `import` and `export`.
- CDN: a public asset delivery server; here where React, ReactDOM, MUI and the fonts are fetched.
- aria-label: the accessibility label attribute on a control element, it lets a screen reader read the correct Vietnamese label.
- TAIL: not used in this prompt.

## 3.BLUEPRINT. Blueprint v4 (verbatim, shared by all 10 prompts)
# BLUEPRINT v4: web chatbot, 10 prompts, DDD, TDD, MUI, first-time API key entry on the page

User lock-in 2026-10-07: the prompt IS the blueprint; an AI that reads all 10 prompts (P0 through P9) can build the chatbot; each prompt is at least 5,000 words, written in formal specification style, covering every case; the first thing to do when the chatbot runs for the first time is TO ASK FOR THE API KEY, send it to the server, the server stores it, and then it runs.

## B1. Goal and first-run flow
- Platform: Windows 11, PowerShell, Python invoked via `py`, coding assistant is opencode, responding model is `gpt-4o-mini` via the OpenAI API (external API).
- First-run flow:
  1. The learner runs `py run.py`; the server starts at `http://127.0.0.1:2610` WHEN THERE IS NO API KEY YET (must not crash).
  2. The browser opens the page; `main.js` calls `GET /api/setup/status`.
  3. It returns `{"configured": false}` ⇒ show the MUI dialog "Nhập khóa API" (password field + Save button).
  4. The learner pastes the key, clicks Save ⇒ `POST /api/setup` with `{"api_key": "<key>"}`.
  5. The server validates the key format, writes it into the `.env` file (line `OPENAI_API_KEY=`) atomically, updates the process memory, returns `{"ok": true}`; it must NEVER return the key.
  6. The page closes the dialog and shows the chat frame; chat runs immediately, no restart needed.
  7. Subsequent runs: `status` returns `{"configured": true}` ⇒ go straight to the chat frame.
- The key lives only in `.env`; not in logs, not in `memory.json`, not in responses, not in errors.

## B4. Domain
- `Message`: frozen dataclass `role: str` (only `"human"` or `"ai"`), `content: str` (non-empty). Wrong ⇒ `ValueError`.
- `trim_history(messages: list[Message], limit: int) -> list[Message]`: returns the LAST `limit` messages, preserving order; `limit <= 0` ⇒ `ValueError`; empty list ⇒ empty; fewer than `limit` ⇒ return all; does not modify the input list.
- `errors.py`: `ApiKeyMissing`, `InvalidApiKey`, `InvalidMessage`, `InvalidSession`, `UpstreamError(status: int, message: str)`; all inherit `ChatbotError(Exception)`.
- Ports (abstract classes `abc.ABC`):
  - `MemoryRepository`: `get(session_id: str) -> list[Message]`, `append(session_id: str, messages: list[Message]) -> None`, `reset(session_id: str) -> None`.
  - `ChatModel`: `reply(system_prompt: str, history: list[Message], user_text: str) -> str`.
  - `ApiKeyStore`: `is_configured() -> bool`, `get() -> str`, `save(api_key: str) -> None`.

## B5. Application (use cases, one function per file)
- `validate_session_id(session_id) -> str`: matches `^[A-Za-z0-9-]{8,64}$`, otherwise `InvalidSession`.
- `validate_message(text) -> str`: trims leading and trailing whitespace; empty ⇒ `InvalidMessage`; longer than 4000 characters ⇒ `InvalidMessage`.
- `validate_api_key(api_key) -> str`: trims whitespace; length 20 to 200; only printable ASCII characters, NO whitespace inside; wrong ⇒ `InvalidApiKey`. Do NOT check the vendor prefix.
- `get_setup_status(store) -> dict`: `{"configured": store.is_configured()}`.
- `configure_api_key(store, api_key) -> None`: `store.save(validate_api_key(api_key))`.
- `send_message(repo, model, store, system_prompt, max_messages, session_id, text) -> str`:
  1. `sid = validate_session_id(...)`, `msg = validate_message(...)`.
  2. `store.is_configured()` false ⇒ `ApiKeyMissing`.
  3. `history = trim_history(repo.get(sid), max_messages)` (if empty, skip trimming).
  4. `reply = model.reply(system_prompt, history, msg)`.
  5. `repo.append(sid, [Message("human", msg), Message("ai", reply)])` ONLY after `reply` exists; a model error ⇒ write NOTHING to memory.
  6. return `reply`.
- `reset_conversation(repo, session_id) -> None`: validate `session_id`, then `repo.reset(sid)`; resetting a non-existent session is a silent success.

## B6. Infrastructure
- `atomic_write_text(path, text)`: writes to a temp file in the same directory then `os.replace`; creates the parent directory if missing; on error ⇒ delete the temp file, rethrow the error.
- `atomic_write_json(path, data)`: `json.dumps(ensure_ascii=False, indent=2)` then `atomic_write_text`.
- `JsonMemoryRepository(path)`: uses `threading.Lock`; file format `{"<session_id>": [{"role": "human", "content": "..."}]}`; missing file ⇒ empty; corrupt file (invalid JSON) ⇒ rename to `memory.json.corrupt-<timestamp>` then start empty, MUST NOT crash; `append` reads then writes atomically within the same lock; stores ALL messages (trimming happens only when reading for the model).
- `load_system_prompt(path) -> str`: reads UTF-8; missing file ⇒ returns a short Vietnamese default, MUST NOT crash; empty file ⇒ also the default.
- `EnvApiKeyStore(env_path)`: `is_configured()` is true exactly when `get()` is non-empty; `get()` reads `OPENAI_API_KEY` from the `.env` file (and from `os.environ` if the file is missing); `save(key)` rewrites the `.env` file keeping the other lines intact, replacing or appending exactly one line `OPENAI_API_KEY=<key>`, writes atomically, updates `os.environ`; write error ⇒ rethrow, MUST NOT leave a half-written file.
- `mask_secret(value) -> str`: returns `"****"` plus the last 4 characters when the length is 12 or more; otherwise `"****"`; used for logs, MUST NEVER write the real key.
- `LangChainChatModel(settings, store)`: each `reply` builds `ChatPromptTemplate.from_messages([("system", system_prompt), MessagesPlaceholder("history"), ("human", "{input}")])`, attaches `ChatOpenAI(model=settings.openai_model, temperature=settings.openai_temperature, timeout=settings.openai_timeout, api_key=store.get())`, converts `Message` to `HumanMessage`/`AIMessage`, returns `.content`; OpenAI error ⇒ `UpstreamError(*map_openai_error(exc))`; the key is read AGAIN on every turn so that no restart is needed after `/api/setup`.
- `map_openai_error(exc) -> tuple[int, str]`: `AuthenticationError`→401 "Khóa API không đúng, nhập lại"; `RateLimitError` (HTTP 429)→503 "Hệ thống đang quá tải, thử lại sau"; `APITimeoutError`→504 "Mô hình trả lời quá lâu"; `APIConnectionError`→502 "Không kết nối được dịch vụ mô hình"; otherwise→500 "Có lỗi không mong đợi".

## 3c. Implementation procedure (TDD)

TERMS, DEFINED ONCE: opencode is a code-writing utility that runs in a command-line window, used here to generate code. An API is an application programming interface, here a service that returns external text. An endpoint is the specific path the API listens on under one HTTP method. A venv is a Python virtual environment, a separate folder holding installed libraries. pytest is the automated testing library for Python. DDD is the layered architecture style organized by business domain, the dependency-arrow rule is in B2. TDD is the discipline of writing in three cycles: write a failing test, write the minimum code to pass, then refactor without breaking green. MUI is the Material Design component set for React. A formal spec is a formal specification, that is, preconditions, postconditions, invariants and error codes written explicitly with logic notation.

PRE: `run.py` and the `src/chatbot/` tree built from P1 to P6 already exist; `py -m pytest -q` runs and is green; the `static/` folder does not exist yet; the file `tests/test_static_rules.py` does not exist yet; the learner's machine browser is a recent Chrome or Edge.

POST: `static/index.html` together with exactly the 10 files in `static/js/` from B2 exist; every file is under 100 lines; no file contains `innerHTML` or `dangerouslySetInnerHTML`; the page opens at `http://127.0.0.1:2610` and runs exactly the B1 flow; every manual test scenario at the end of section 3c has an observable result to record in the P7 report.

INV: each file in `static/` keeps exactly one concern; each function lives in exactly one file; content is always a CHILD STRING of a React element; the API key never touches browser storage; every button and every input has an `aria-label`; only create files that are in the B2 tree, do not generate any additional helper files outside that list; ∀ `fetch` call ⇒ returns an object with exactly the three keys `ok`, `status`, `data`, that is, no call throws an exception out of the file; ∀ displayed message ⇒ the content goes through the `item.content` of the message, passed as a child of the `Typography` component, not through a raw string write mechanism.

INTERPRETATION NOTE (matches B8): a file is one download unit for the browser, so the nine sub-files use classic `<script>` and register into `window.Chatbot`; `main.js` is loaded with `<script type="module">` and uses no `import` or `export` statement, so the file is both exactly an ES module and syntax-checkable with Node without depending on file mode. Load order is a mandatory rule: the nine sub-files first, `main.js` after. On the first run, `SetupDialog` is built with `open=true` before the user touches the chat frame; because it is a modal MUI `Dialog` it blocks the whole screen, and the chat frame is only a background layer. After the key is saved successfully, the dialog closes and the chat frame accepts interaction. In every file under `static/`, the three globals `React`, `ReactDOM`, `MaterialUI` are read inside functions or at the initialization block, not read at module level, so one broken file does not kill the others. Finally, no file in `static/` is allowed to be compiled, so all the code is plain `React.createElement`.

from pathlib import Path

import pytest

STATIC = Path(__file__).resolve().parents[1] / "static"
JS = ["js/session_id.js", "js/api_status.js", "js/api_setup.js", "js/api_chat.js",
      "js/api_reset.js", "js/message_bubble.js", "js/setup_dialog.js",

FORBID = ["innerHTML", "dangerouslySetInnerHTML", "document.write", "eval(", "require("]
LIMIT = 100
NET = ["fetch(", "ok: false", "status: 0", "Không kết nối được máy chủ"]
RULES = [

def test_03_no_forbidden_token():
    hits = []
    for path in sorted(STATIC.rglob("*")):
        if path.is_file() and path.suffix in {".js", ".html"}:
            text = path.read_text(encoding="utf-8")
            hits += [f"{path.name}:{token}" for token in FORBID if token in text]
    assert hits == [], f"cấm dùng: {hits}"


def test_04_index_html_shape():
    text = body("index.html")

    assert text.index("js/main.js") > text.index("js/chat_view.js"), "sai thứ tự nạp"

def test_05_markers(name, markers):
    text = body(name)
    missing = [m for m in markers if not m.startswith("!") and m not in text]
    banned = [m[1:] for m in markers if m.startswith("!") and m[1:] in text]
    assert missing == [], f"{name} thiếu {missing}"
    assert banned == [], f"{name} cấm {banned}"


@pytest.mark.parametrize("name", VIEW)
def test_06_component_uses_create_element(name):
    assert "React.createElement" in body(name), name

// Mã phiên dùng chung cho mọi lượt gửi, theo B8. Không đụng khóa API.
const CB = (window.Chatbot = window.Chatbot || {});
const PATTERN = /^[A-Za-z0-9-]{8,64}$/;

function isValid(value) {
  return typeof value === "string" && PATTERN.test(value);
}

function randomPart() {
  return Math.floor(Math.random() * 0x100000000).toString(36).padStart(6, "0");
}

function fallbackId() {
  return randomPart() + randomPart() + randomPart() + randomPart();
}

function readStored() {
  try {
    return window.localStorage.getItem("session_id");
  } catch (err) {
    return null;
  }
}

function writeStored(value) {
  try {
    window.localStorage.setItem("session_id", value);
  } catch (err) {
    // Bộ nhớ trình duyệt bị chặn: mã phiên vẫn dùng được trong RAM của trang.
  }
}

function newId() {
  const hasUuid = window.crypto && typeof window.crypto.randomUUID === "function";
  const created = hasUuid ? window.crypto.randomUUID() : fallbackId();
  return isValid(created) ? created : fallbackId();
}

function getSessionId() {
  const stored = readStored();
  if (isValid(stored)) {
    return stored;
  }
  const created = newId();
  writeStored(created);
  return created;
}

function isValidSessionId(value) {
  return isValid(value);
}

CB.sessionId = { getSessionId: getSessionId, isValidSessionId: isValidSessionId };
```

// Gửi tin và xóa lịch sử. Chặn bấm Gửi hai lần bằng cờ sending ở cấp mô đun.
const CB = (window.Chatbot = window.Chatbot || {});

let sending = false;

async function sendMessage(options) {
  if (sending || options.text.trim() === "") {
    return false;
  }
  sending = true;
  options.setBusy(true);
  options.setError("");
  options.append({ role: "human", content: options.text.trim() });
  try {
    const result = await CB.apiChat.apiChat(options.sessionId, options.text.trim());
    if (result.status === 0) {
      options.setError(result.data.message);
      return false;
    }
    if (!result.ok) {
      if (result.status === 409) {
        options.onRequireKey("Chưa có khóa API, hãy nhập khóa rồi thử lại");
        return false;
      }
      if (result.status === 401) {
        options.onRequireKey("Khóa API không đúng, nhập lại");
        return false;
      }
      options.setError(result.data.message || "Không nhận được câu trả lời");
      return false;
    }
    options.append({ role: "ai", content: result.data.reply });
    return true;
  } finally {
    sending = false;
    options.setBusy(false);
  }
}

async function resetConversation(options) {
  const result = await CB.apiReset.apiReset(options.sessionId);
  if (result.status === 0) {
    options.setError(result.data.message);
    return false;
  }
  if (!result.ok) {
    options.setError(result.data.message || "Không xóa được lịch sử");
    return false;
  }
  options.clear();
  return true;
}

CB.send = { sendMessage: sendMessage, resetConversation: resetConversation };

const CB = (window.Chatbot = window.Chatbot || {});

function showFatal(text) {
  document.getElementById("root").textContent = text;
}

function App() {
  const [phase, setPhase] = React.useState("boot");
  const [messages, setMessages] = React.useState([]);
  const [draft, setDraft] = React.useState("");
  const [busy, setBusy] = React.useState(false);

    setPhase("ready");
    return { ok: true };

    }

      sessionId: CB.sessionId.getSessionId(),
      clear: () => setMessages([]),
      setError: setError,
    });

  }
  if (phase === "broken") {

      React.createElement(MaterialUI.Alert, { severity: "warning", sx: { mb: 2 } }, error),
      React.createElement(MaterialUI.Button, { variant: "contained", onClick: start }, "Thử lại"));

      onDraftChange: setDraft, onSend: send, onReset: reset,

## ERR

ERR.1 Symptom: blank page, only the line `Đang tải giao diện, vui lòng chờ...` remains. Cause: the CDN is blocked by a corporate firewall or the network is down, so `window.React` or `window.MaterialUI` is missing. Fix: the `else` branch of `main.js` calls `showFatal`, using `textContent` to write a Vietnamese message that clearly names the CDN path that is failing to load and suggests reloading the page; do NOT redraw the UI yourself. Related files: `static/index.html`, `static/js/main.js`.

ERR.2 Symptom: `main.js` runs before `window.Chatbot` has all its functions, so `CB.sessionId` is `undefined`. Cause: wrong order of the `script` tags in `index.html`. Fix: it is mandatory to load the nine sub-files first and `main.js` after; `test_04_index_html_shape` checks this order by comparing string positions. Related files: `static/index.html`, `tests/test_static_rules.py`.

ERR.3 Symptom: every page reload generates a new session id and the previous history disappears. Cause: `getSessionId` returns `null` from `readStored` because the key is stored under the wrong name. Fix: use exactly one key name constant `session_id` for both reading and writing. Related file: `static/js/session_id.js`.

ERR.4 Symptom: the session id does not match `^[A-Za-z0-9-]{8,64}$` so the server returns `invalid_session`. Cause: `crypto.randomUUID` is unavailable because the page is open in an insecure context, and the fallback generates a string that is too short. Fix: if the generated id is invalid then regenerate with `fallbackId`; re-check with `isValid` before returning. Related file: `static/js/session_id.js`.

ERR.5 Symptom: `fetch` throws `TypeError` and the dialog reports a network error although the server is still running. Cause: the server is off or the port is wrong. Fix: every network call file wraps `try/catch` and returns `{ok: false, status: 0, data: {message: "Không kết nối được máy chủ"}}`; `main.js` switches to `phase = "broken"` with a `Thử lại` button. Related files: `static/js/api_status.js`, `static/js/api_chat.js`, `static/js/main.js`.

ERR.6 Symptom: a network error makes the page jump back to the API key dialog even though a key exists. Cause: the dialog-opening condition relies on `status` instead of on `configured`. ERR.7 Symptom: the API key appears in browser storage after saving. Cause: the key state is written down to the page's temporary storage. Fix: the key lives only in `useState` and goes straight into `fetch`; the two negative markers `localStorage` and `sessionStorage` in `test_05` block the regression. Related files: `static/js/setup_dialog.js`, `tests/test_static_rules.py`.

ERR.8 Symptom: after clicking `Lưu`, the key field still holds the old characters. Cause: the clearing command after obtaining a result is missing. Related file: `static/js/setup_dialog.js`.

ERR.9 Symptom: clicking `Lưu` many times creates many `/api/setup` requests. Cause: there is no in-flight blocking flag. Fix: `submit()` exits early if `busy` is on, together with `disabled: busy` on the `Lưu` button and on the input. Related file: `static/js/setup_dialog.js`.

ERR.10 Symptom: typing two messages in quick succession shows two human messages in the chat frame but only one AI message. Cause: each round overwrites the list instead of appending. Fix: use the functional update `setMessages((old) => old.concat([item]))`; the `sending` flag blocks two overlapping rounds. Related files: `static/js/main.js`, `static/js/send_handler.js`.

ERR.11 Symptom: clicking `Gửi` twice creates two `/api/chat` requests. Cause: React's `busy` state has not updated in time between the two clicks in the same beat. // Mã phiên cho một tab trình duyệt. An toàn khi localStorage bị chặn.
let duPhong = null;

function sinhMa() {
  if (window.crypto && typeof window.crypto.randomUUID === "function") {
    return window.crypto.randomUUID();
  }
  const bang = "abcdefghijklmnopqrstuvwxyz0123456789";
  let ket = "";
  for (let i = 0; i < 24; i += 1) {
    ket += bang.charAt(Math.floor(Math.random() * bang.length));
  }
  return ket;
}

export function getSessionId() {
  let kho = null;
  try {
    kho = window.localStorage;
  } catch (loi) {
    kho = null;
  }
  if (kho === null) {
    if (duPhong === null) { duPhong = sinhMa(); }
    return duPhong;
  }
  let gia = null;
  try { gia = kho.getItem("session_id"); } catch (loi) { gia = null; }
  if (gia !== null && /^[A-Za-z0-9-]{8,64}$/.test(gia)) { return gia; }
  if (duPhong === null) { duPhong = sinhMa(); }
  try { kho.setItem("session_id", duPhong); } catch (loi) { /* chỉ giữ trong bộ nhớ */ }
  return duPhong;
}

## 5.10. `static/js/send_handler.js`

```javascript
// Luồng gửi một tin: thêm tin người, gọi API, thêm tin AI hoặc thông báo lỗi.
import { apiChat } from "./api_chat.js";

const SAI_KHOA = "Khóa API không đúng. Mở hộp thoại để nhập lại.";
const THIEU_KHOA = "Chưa có khóa API trên máy chủ. Hãy nhập khóa rồi thử lại.";

export function taoSendHandler(cu) {
  // cu = { docTrang, ghiTrang, moHopKhoa }
  const themTin = function (role, content) {
    const s = cu.docTrang();
    cu.ghiTrang({ tin: s.tin.concat([{ role: role, content: content }]) });
  };

  return async function gui(text) {
    if (cu.docTrang().dangGui) { return { ok: false }; }
    cu.ghiTrang({ dangGui: true, loi: "" });
    themTin("human", text);
    const ket = await apiChat(cu.docTrang().sessionId, text);
    if (ket.ok && typeof ket.data.reply === "string" && ket.data.reply.length > 0) {
      themTin("ai", ket.data.reply);
      cu.ghiTrang({ dangGui: false, loi: "" });
      return { ok: true };
    }
    cu.ghiTrang({ dangGui: false });
    if (ket.status === 409) {
      cu.moHopKhoa(THIEU_KHOA);
      return { ok: false };
    }
    if (ket.status === 401) {
      cu.moHopKhoa(SAI_KHOA);
      return { ok: false };
    }
    cu.ghiTrang({ loi: ket.data.message || "Không gửi được tin nhắn." });
    return { ok: false };
  };
}
```

  const boNho = React.useRef({ tin: [], dangGui: false, sessionId: getSessionId() }).current;

  const ghi = React.useCallback(function (moi) {
    Object.assign(boNho, moi);
    setUi(function (c) { return Object.assign({}, c, moi); });
  }, []);

  const moHop = React.useCallback(function (thongDiep) {
    setUi(function (c) {
      return Object.assign({}, c, { hop: true, loiHop: thongDiep || "", token: c.token + 1 });
    });
  }, []);

  const guiTin = React.useMemo(function () {
    return taoSendHandler({
      docTrang: function () { return boNho; },
      ghiTrang: ghi,
      moHopKhoa: moHop
    });
  }, [ghi, moHop]);

  const kiemTra = React.useCallback(function () {
    apiStatus().then(function (ket) {
      const loi = ket.status === 0
        ? "Không kết nối được máy chủ. Chạy py run.py rồi bấm Thử lại."
        : (ket.ok ? "" : "Máy chủ trả lỗi " + ket.status + ": "
          + (ket.data.message || "không rõ nguyên nhân"));
      setUi(function (c) {
        return { tai: true, hop: ket.ok && ket.data.configured !== true, loiHop: "",
                 loi: "", loiMayChu: loi, token: c.token + 1 };
      });
    });
  }, []);

  React.useEffect(function () { kiemTra(); }, [kiemTra]);

  const chatMoi = React.useCallback(function () {
    apiReset(boNho.sessionId).then(function (ket) {
      if (ket.ok) { ghi({ tin: [], loi: "" }); return; }
      ghi({ loi: ket.data.message || "Không xoá được cuộc trò chuyện." });
    });
  }, [ghi]);

  if (!ui.tai) {
    return el(Box, { sx: { height: "100vh", display: "flex", alignItems: "center",
                            justifyContent: "center" } },
      el(CircularProgress, { "aria-label": "Đang kiểm tra trạng thái máy chủ" }));
  }

  if (ui.loiMayChu !== "") {
    return el(Box, { sx: { height: "100vh", display: "flex", flexDirection: "column", gap: 2,
                            alignItems: "center", justifyContent: "center", p: 3 } },
      el(Alert, { severity: "error", "aria-label": "Lỗi kết nối máy chủ" }, ui.loiMayChu),
      el(Button, {
        variant: "contained", "aria-label": "Nút thử lại kết nối",
        onClick: function () {
          setUi(function (c) { return Object.assign({}, c, { loiMayChu: "" }); });
          kiemTra();
        }
      }, "Thử lại"));
  }

  return el(React.Fragment, null,
    el(SetupDialog, {
      mo: ui.hop, loiKhoiDau: ui.loiHop, token: ui.token,
      onSaved: function () { ghi({ hop: false, loiHop: "" }); },
      onClose: function () { kiemTra(); }
    }),
    ui.hop ? null : el(ChatView, {
      tin: boNho.tin, dangGui: boNho.dangGui, loi: ui.loi,
      onSend: guiTin, onReset: chatMoi
    }));
}

ReactDOM.createRoot(document.getElementById("root")).render(el(App));
